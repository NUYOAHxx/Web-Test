# ============================================================================
# Unitree 风格通用人形机器人 —— Python 依赖清单
# ----------------------------------------------------------------------------
# 版本约束经过实机验证, 组合具有强耦合性, 请勿随意放宽:
#
#   1. pin (Pinocchio) 与 coal 必须成对安装。
#      pin 3.9.0 要求 cmeel-boost~=1.89, 而 coal 3.0.3 要求 cmeel-boost~=1.90,
#      两者互斥; 唯一自洽组合是 pin==3.9.0 + coal==3.0.2 (均为 cmeel-boost~=1.89)。
#      若只写 `pin` 让解析器自由选版, 大概率落到 pin 4.x + coal 3.0.3,
#      此时 API 与本工程代码不匹配。
#
#   2. 必须安装 `pin-pink`, 不是 `pink`。
#      PyPI 上的 `pink` 是一个同名的代码格式化工具 (抢注), 安装后 `import pink`
#      会抛 ImportError, 而非 ModuleNotFoundError, 极易误判。
#      Inria Pink 的发行名是 pin-pink, 安装后模块名才是 pink。
#
#   3. cmeel-boost 1.89 强制 numpy>=2.3。
#      因此 scipy / matplotlib 必须同步升级到支持 numpy 2 的版本,
#      发行版自带的旧版本 (scipy 1.11 / matplotlib 3.6, 均为 numpy 1.x 编译)
#      在 numpy 2 下会直接崩溃。
# ============================================================================

# ── 刚体动力学与几何 ───────────────────────────────────────────────────────
pin==3.9.0            # PyPI 包名 pin, import 名 pinocchio
coal==3.0.2           # Pinocchio 3.9 对应的碰撞几何后端 (须与 pin 严格配套)

# ── 逆运动学求解器 ─────────────────────────────────────────────────────────
pin-pink>=4.4.0       # 注意: 不是 pink
proxsuite>=0.7.3      # Pink 底层的凸二次规划求解器 (ProxQP)

# ── 数值计算 ──────────────────────────────────────────────────────────────
numpy>=2.3,<2.4       # 被 cmeel-boost 强制; 上界随其演进
scipy>=1.15           # 必须支持 numpy 2
qpsolvers>=4.13       # 双臂求解器的稀疏 QP 接口

# ── 物理仿真 (可选: 仅 simulation/deploy_mujoco53.py 需要) ─────────────────
mujoco>=3.15
# torch>=2.0          # 加载 simulation/policy/policy.pt 需要, 体积大故不默认安装

# ── 可视化与工具 ──────────────────────────────────────────────────────────
matplotlib>=3.10      # 三维规划基准绘图, 须支持 numpy 2
pyyaml>=6.0

# ── 测试 ──────────────────────────────────────────────────────────────────
pytest>=7.4
